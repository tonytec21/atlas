SET NAMES utf8mb4;
DROP DATABASE IF EXISTS atlas;
CREATE DATABASE atlas DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE atlas;
CREATE TABLE tarefas (
  id INT AUTO_INCREMENT PRIMARY KEY, token VARCHAR(64), titulo VARCHAR(255),
  categoria VARCHAR(20), origem VARCHAR(20), descricao TEXT, data_limite DATETIME NULL,
  funcionario_responsavel VARCHAR(255), criado_por VARCHAR(255), data_criacao DATETIME NULL,
  data_atualizacao DATETIME NULL, atualizado_por VARCHAR(255), caminho_anexo TEXT,
  nivel_de_prioridade VARCHAR(50), revisor VARCHAR(255), status VARCHAR(80) DEFAULT 'Em Andamento',
  data_conclusao DATETIME NULL, sub_categoria VARCHAR(10) NULL, id_tarefa_principal INT NULL,
  numero_oficio VARCHAR(50) NULL) ENGINE=InnoDB;
CREATE TABLE funcionarios (id INT AUTO_INCREMENT PRIMARY KEY, nome_completo VARCHAR(255),
  usuario VARCHAR(100), senha VARCHAR(255), nivel_de_acesso VARCHAR(50), acesso_adicional TEXT,
  status VARCHAR(20) DEFAULT 'Ativo') ENGINE=InnoDB;
CREATE TABLE categorias (id INT AUTO_INCREMENT PRIMARY KEY, titulo VARCHAR(255), status VARCHAR(20)) ENGINE=InnoDB;
CREATE TABLE origem (id INT AUTO_INCREMENT PRIMARY KEY, titulo VARCHAR(255), status VARCHAR(20)) ENGINE=InnoDB;
CREATE TABLE comentarios (id INT AUTO_INCREMENT PRIMARY KEY, hash_tarefa VARCHAR(64), comentario TEXT,
  caminho_anexo TEXT, data_comentario DATETIME NULL, data_atualizacao DATETIME NULL,
  funcionario VARCHAR(255), status VARCHAR(20), id_tarefa_principal INT NULL) ENGINE=InnoDB;
CREATE TABLE recibos_de_entrega (id INT AUTO_INCREMENT PRIMARY KEY, task_id INT, receptor VARCHAR(255), data_entrega DATETIME NULL) ENGINE=InnoDB;
CREATE TABLE guia_de_recebimento (id INT AUTO_INCREMENT PRIMARY KEY, task_id INT, cliente VARCHAR(255), data_recebimento DATETIME NULL) ENGINE=InnoDB;
CREATE TABLE oficios (id INT AUTO_INCREMENT PRIMARY KEY, numero VARCHAR(50), assunto VARCHAR(255)) ENGINE=InnoDB;
CREATE TABLE cadastro_serventia (id INT AUTO_INCREMENT PRIMARY KEY, nome VARCHAR(255)) ENGINE=InnoDB;
CREATE TABLE logs_tarefas (id INT AUTO_INCREMENT PRIMARY KEY, descricao TEXT) ENGINE=InnoDB;

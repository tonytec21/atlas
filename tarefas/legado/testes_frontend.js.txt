function checar(desc, cond, detalhe) {
    if (cond) { ok++; console.log('  [OK]    ' + desc); }
    else { falhas.push(desc + (detalhe ? ' -> ' + detalhe : '')); console.log('  [FALHA] ' + desc + (detalhe ? ' -> ' + detalhe : '')); }
}

// Teste 1 — recursão da zona de upload
// ---------------------------------------------------------------
console.log('\n== 1. Zona de anexos (RangeError) ==');
{
    const d = window.document;
    const zona = d.createElement('div');
    zona.id = 'zonaUpload';
    const input = d.createElement('input');
    input.type = 'file';
    input.id = 'attachments';
    zona.appendChild(input);
    d.body.appendChild(zona);

    // Reproduz o padrão ANTIGO, para provar que ele estourava a pilha.
    let estourou = false;
    $('#zonaUpload').on('click', function () { $('#attachments').trigger('click'); });
    try { $('#zonaUpload').trigger('click'); }
    catch (e) { estourou = /call stack|too much recursion/i.test(e.message); }
    checar('padrão antigo realmente estoura a pilha', estourou);

    // Agora o padrão CORRIGIDO.
    $('#zonaUpload').off('click');
    let cliquesNoInput = 0;
    $('#zonaUpload').on('click', function (e) {
        if (e.target === d.getElementById('attachments')) { return; }
        cliquesNoInput++;
        if (cliquesNoInput > 50) { throw new Error('recursão'); }
        $('#attachments').trigger('click');
    });

    let erro = null;
    try { $('#zonaUpload').trigger('click'); } catch (e) { erro = e.message; }
    checar('padrão corrigido não entra em recursão', erro === null, erro);
    checar('padrão corrigido abre o seletor uma única vez', cliquesNoInput === 1, 'cliques=' + cliquesNoInput);

    zona.remove();
}

const chamadas = [];
let atrasoProximaChamada = 0;
$.ajax = function (opcoes) {
    const url = opcoes.url || '';
    const dados = opcoes.data || {};
    chamadas.push({ url, visao: dados.visao });

    let corpo;
    if (url.indexOf('tarefas.php') !== -1) {
        corpo = dados.visao === 'kanban' ? RESP.kanban
              : dados.visao === 'calendario' ? RESP.calendario : RESP.cards;
    } else if (url.indexOf('dashboard.php') !== -1) { corpo = RESP.dashboard; }
    else if (url.indexOf('tarefa.php') !== -1) { corpo = RESP.detalhe; }
    else { corpo = { success: true }; }

    const atraso = atrasoProximaChamada;
    atrasoProximaChamada = 0;
    const d = $.Deferred();
    setTimeout(() => d.resolve(corpo), atraso);
    return d.promise();
};
$.fn.modal = function () { return this; };

function titulo($t) { echo "\n\033[1m== $t ==\033[0m\n"; }

function checar($descricao, $condicao, $detalhe = '')
{
    global $ok, $falhas;
    if ($condicao) {
        $ok++;
        echo "  [OK]    $descricao\n";
    } else {
        $falhas[] = $descricao . ($detalhe ? " -> $detalhe" : '');
        echo "  [FALHA] $descricao" . ($detalhe ? " -> $detalhe" : '') . "\n";
    }
}

/** Executa um endpoint em processo isolado e devolve a saída decodificada. */
function chamar($arquivo, $get = array(), $post = array(), $usuario = 'tony')
{
    global $RAIZ;

    $script = tempnam(sys_get_temp_dir(), 'tf') . '.php';
    $codigo = '<?php' . "\n"
        . '$_GET = ' . var_export($get, true) . ";\n"
        . '$_POST = ' . var_export($post, true) . ";\n"
        . '$_REQUEST = array_merge($_GET, $_POST);' . "\n"
        . '$_SERVER["REQUEST_METHOD"] = ' . var_export($post ? 'POST' : 'GET', true) . ";\n"
        . '$_SERVER["HTTP_X_REQUESTED_WITH"] = "XMLHttpRequest";' . "\n"
        . 'session_start();' . "\n"
        . '$_SESSION["username"] = ' . var_export($usuario, true) . ";\n"
        . '$_SESSION["tarefas_csrf"] = "teste";' . "\n"
        . '$_POST["_csrf"] = "teste"; $_GET["_csrf"] = "teste"; $_REQUEST["_csrf"] = "teste";' . "\n"
        . 'require ' . var_export($RAIZ . '/' . $arquivo, true) . ";\n";

    file_put_contents($script, $codigo);
    $saida = shell_exec('php ' . escapeshellarg($script) . ' 2>&1');
    @unlink($script);

    $json = json_decode((string) $saida, true);
    return $json === null ? array('__raw' => trim((string) $saida)) : $json;
}
